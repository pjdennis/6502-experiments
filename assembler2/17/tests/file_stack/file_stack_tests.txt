# File Stack Component Tests
#
# Format:
#   NAME: test identifier
#   MODE: echo|lines|info|memory
#   FILE <filename>: (multi-line content, "N: " prefix stripped)
#   EXPECT_STDOUT: (multi-line expected output)
#   EXPECT_STDERR: (multi-line expected stderr, for error tests)
#
# The first FILE defined is the main input file.
# For memory mode, use @include <filename> to include other files.

# ============================================================================
# ECHO MODE - Basic character reading
# ============================================================================

---
NAME: echo_simple
MODE: echo
FILE input.txt:
 1: Hello
EXPECT_STDOUT:
Hello

---
NAME: echo_multiline
MODE: echo
FILE input.txt:
 1: Line1
 2: Line2
 3: Line3

Line2
Line3

FILE input.txt:
EXPECT_STDOUT:

---
NAME: echo_no_trailing_newline

After

# ============================================================================
# TRACEBACK MODE - Stack traceback testing
# ============================================================================
# Tests @traceback which prints the file stack and pops all entries

---
NAME: trace_simple
MODE: memory
FILE main.txt:
 1: Before
 2: @traceback
 3: After
EXPECT_STDOUT:
Before
file:main.txt:2

# Note: After @traceback pops the stack, reading continues but finds EOF
# so "After" is never printed

---
NAME: trace_nested
MODE: memory
FILE main.txt:
 1: Main start
 2: @include inner.txt
 3: Main end
FILE inner.txt:
 1: Inner start
 2: @traceback
 3: Inner end
EXPECT_STDOUT:
Main start
Inner start
file:inner.txt:2
file:main.txt:2

---
NAME: trace_deep
MODE: memory
FILE main.txt:
 1: Outer1
 2: @include middle.txt
 3: Outer2
FILE middle.txt:
 1: Middle1
 2: @include inner.txt
 3: Middle2
FILE inner.txt:
 1: Inner
 2: @traceback
EXPECT_STDOUT:
Outer1
Middle1
Inner
file:inner.txt:2
file:middle.txt:2
file:main.txt:2

---
NAME: trace_with_memory
MODE: memory
FILE main.txt:
 1: File start
 2: @memory MEM_CONTENT
 3: @traceback
 4: After trace
EXPECT_STDOUT:
File start
MEM_CONTENT
file:main.txt:2

# Note: Line shows 2 (where @memory was) because after memory exhausts,
# we restore the line from when memory was pushed

---
NAME: trace_alternating
MODE: memory
FILE main.txt:
 1: Main line 1
 2: @include inner.txt
 3: Main line 3
FILE inner.txt:
 1: Inner line 1
 2: @memory @traceback
 3: Inner line 3
EXPECT_STDOUT:
Main line 1
Inner line 1
memory:MEMORY:1
file:inner.txt:2
file:main.txt:2

# This tests file -> file -> memory stack with traceback showing all types
# The @traceback is inside the memory content, so it triggers while memory source is active

# ============================================================================
# FRAMES MODE - Frame chain inspection
# ============================================================================
# Tests the @frames directive (frames mode only) which non-destructively
# walks the stack and prints "depth:type:name" for each frame, top-of-stack
# first. Frames mode otherwise behaves like memory mode.

---
NAME: frames_single
MODE: frames
FILE input.txt:
 1: Before
 2: @frames
 3: After
EXPECT_STDOUT:
Before
0:file:input.txt
After

---
NAME: frames_with_include
MODE: frames
FILE main.txt:
 1: Top
 2: @include inner.txt
 3: End
FILE inner.txt:
 1: Inner
 2: @frames
EXPECT_STDOUT:
Top
Inner
0:file:inner.txt
1:file:main.txt
End

---
NAME: frames_with_memory
MODE: frames
FILE input.txt:
 1: A
 2: @memory @frames
 3: B
EXPECT_STDOUT:
A
0:memory:MEMORY
1:file:input.txt
B

---
NAME: frames_deep
MODE: frames
FILE main.txt:
 1: M1
 2: @include middle.txt
 3: M3
FILE middle.txt:
 1: D1
 2: @include inner.txt
 3: D3
FILE inner.txt:
 1: I1
 2: @frames
EXPECT_STDOUT:
M1
D1
I1
0:file:inner.txt
1:file:middle.txt
2:file:main.txt
D3
M3

# ============================================================================
# PUSH/POP BALANCE - Verify FS_P16 returns to its starting value
# ============================================================================
# These tests use @frames at multiple points to assert that the source stack
# unwinds correctly. Each @frames line reports depth:type:name; balance is
# proven when the post-pop @frames matches the pre-push @frames.

---
NAME: balance_after_include
MODE: frames
FILE main.txt:
 1: @frames
 2: @include inner.txt
 3: @frames
FILE inner.txt:
 1: inner content
EXPECT_STDOUT:
0:file:main.txt
inner content
0:file:main.txt

---
NAME: balance_after_memory
MODE: frames
FILE input.txt:
 1: @frames
 2: @memory MEMDATA
 3: @frames
EXPECT_STDOUT:
0:file:input.txt
MEMDATA
0:file:input.txt

---
NAME: balance_repeated_include
MODE: frames
FILE main.txt:
 1: @include inner.txt
 2: @include inner.txt
FILE inner.txt:
 1: @frames
EXPECT_STDOUT:
0:file:inner.txt
1:file:main.txt
0:file:inner.txt
1:file:main.txt

---
NAME: balance_different_name_lengths
MODE: frames
FILE main.txt:
 1: @include a.txt
 2: @include longer_name_file.txt
 3: @frames
FILE a.txt:
 1: @frames
FILE longer_name_file.txt:
 1: @frames
EXPECT_STDOUT:
0:file:a.txt
1:file:main.txt
0:file:longer_name_file.txt
1:file:main.txt
0:file:main.txt

---
NAME: balance_alternating_include_memory
MODE: frames
FILE main.txt:
 1: @include inner.txt
 2: @memory MEM
 3: @frames
FILE inner.txt:
 1: @frames
EXPECT_STDOUT:
0:file:inner.txt
1:file:main.txt
MEM
0:file:main.txt

---
NAME: balance_memory_then_include
MODE: frames
FILE main.txt:
 1: @memory FIRST
 2: @include inner.txt
 3: @frames
FILE inner.txt:
 1: @frames
EXPECT_STDOUT:
FIRST
0:file:inner.txt
1:file:main.txt
0:file:main.txt

---
NAME: balance_nested_with_memory
MODE: frames
FILE main.txt:
 1: @include inner.txt
 2: @frames
FILE inner.txt:
 1: @memory NESTED
 2: @frames
EXPECT_STDOUT:
NESTED
0:file:inner.txt
1:file:main.txt
0:file:main.txt
